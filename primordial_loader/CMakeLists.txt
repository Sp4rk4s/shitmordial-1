cmake_minimum_required(VERSION 3.20)
project(primordial_loader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure primordial_loader with -A Win32; the payload is x86.")
endif()
add_subdirectory(../third_party/minhook "${CMAKE_CURRENT_BINARY_DIR}/minhook")
# Explicitly injected diagnostic preview; never loaded by either production DLL.
option(BUILD_MENU_PREVIEW "Build a standalone diagnostic DLL using the production menu code" OFF)
option(BUILD_MENU_LAYOUT_PROBES "Build historical incremental menu-layout probes" OFF)
if(BUILD_MENU_PREVIEW)
add_library(menu_preview SHARED src/menu_preview_entry.cpp src/menu_ui.cpp src/menu_profile.cpp)
# Keep the SEH diagnostic boundary out of the preview compiler's optimizer.
target_compile_options(menu_preview PRIVATE /Od)
target_compile_definitions(menu_preview PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(menu_preview PRIVATE minhook gdi32 user32 bcrypt)
endif()
if(BUILD_MENU_LAYOUT_PROBES)
add_library(menu_preview_layout SHARED src/menu_preview_layout.cpp)
target_compile_options(menu_preview_layout PRIVATE /Od)
target_compile_definitions(menu_preview_layout PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(menu_preview_layout PRIVATE minhook)
add_library(menu_preview_content SHARED src/menu_preview_layout.cpp)
target_compile_options(menu_preview_content PRIVATE /Od)
target_compile_definitions(menu_preview_content PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CONTENT_LAYOUT_PROBE)
target_link_libraries(menu_preview_content PRIVATE minhook)
endif()
add_library(primordial_tls SHARED src/tls_carrier.cpp src/tls_carrier.def)
target_compile_options(primordial_tls PRIVATE /GS-)
target_link_libraries(primordial_tls PRIVATE kernel32)
target_link_options(primordial_tls PRIVATE /NODEFAULTLIB /ENTRY:DllMain@12 /INCLUDE:__tls_used)
# Steam embeds and stages the binary; this DLL patches and initializes it.
add_library(${PROJECT_NAME} SHARED src/dllmain.cpp src/mapper.cpp src/pe_builder.cpp
    src/crashlog.cpp src/signatures.cpp src/pattern_hook.cpp src/imports.cpp src/startup.cpp src/tls.cpp src/game_modules.cpp
    src/menu_ui.cpp src/menu_profile.cpp src/frame_stage.cpp)
# MSVC19.52 optimizer crashes on the nested native-SEH paint diagnostic boundary.
set_source_files_properties(src/menu_ui.cpp PROPERTIES COMPILE_OPTIONS /Od)
add_dependencies(primordial_loader primordial_tls)
target_include_directories(${PROJECT_NAME} PRIVATE src)
target_compile_definitions(${PROJECT_NAME} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(${PROJECT_NAME} PRIVATE dbghelp minhook gdi32 user32 bcrypt)
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/minhook/LICENSE.txt"
        "$<TARGET_FILE_DIR:primordial_loader>/MinHook-LICENSE.txt")
include(CTest)
if(BUILD_TESTING)
    add_executable(menu_input_tests ../tests/menu_input_tests.cpp)
    add_test(NAME menu-input-transitions COMMAND menu_input_tests)
    add_executable(menu_profile_tests ../tests/menu_profile_tests.cpp src/menu_profile.cpp)
    target_compile_definitions(menu_profile_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(menu_profile_tests PRIVATE bcrypt)
    add_test(NAME menu-source-profile COMMAND menu_profile_tests)
    set_tests_properties(menu-source-profile PROPERTIES TIMEOUT 30)
    add_executable(crashlog_tests ../tests/crashlog_tests.cpp src/crashlog.cpp)
    target_link_libraries(crashlog_tests PRIVATE dbghelp)
    target_compile_definitions(crashlog_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(crashlog_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/crashlog probe")
    add_test(NAME exception-notifications COMMAND crashlog_tests)
    set_tests_properties(exception-notifications PROPERTIES TIMEOUT 30)
    add_executable(tls_tests ../tests/tls_tests.cpp src/tls.cpp)
    add_dependencies(tls_tests primordial_tls)
    target_compile_definitions(tls_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_test(NAME static-tls COMMAND tls_tests "${CMAKE_CURRENT_SOURCE_DIR}/../primordial.bin")
    set_tests_properties(static-tls PROPERTIES TIMEOUT 30)
    add_executable(startup_tests ../tests/startup_tests.cpp src/startup.cpp src/game_modules.cpp)
    add_test(NAME captured-crt-startup COMMAND startup_tests "${CMAKE_CURRENT_SOURCE_DIR}/../primordial.bin")
    set_tests_properties(captured-crt-startup PROPERTIES TIMEOUT 30)
    add_executable(import_tests ../tests/import_tests.cpp src/imports.cpp)
    target_compile_definitions(import_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_test(NAME import-mapping COMMAND import_tests "${CMAKE_CURRENT_SOURCE_DIR}/../primordial.bin")
    set_tests_properties(import-mapping PROPERTIES TIMEOUT 30)
    add_executable(pattern_hook_tests ../tests/pattern_hook_tests.cpp src/pattern_hook.cpp)
    target_link_libraries(pattern_hook_tests PRIVATE minhook)
    target_compile_definitions(pattern_hook_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_test(NAME pattern-hook COMMAND pattern_hook_tests)
    set_tests_properties(pattern-hook PROPERTIES TIMEOUT 30)
    add_executable(mapping_tests ../tests/mapping_tests.cpp ../steam_module/staging.cpp
        ../steam_module/iat_hook.cpp src/mapper.cpp src/pe_builder.cpp)
    target_include_directories(mapping_tests PRIVATE src)
    target_compile_definitions(mapping_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_custom_command(TARGET mapping_tests POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:mapping_tests>/mapping probe"
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:mapping_tests>" "$<TARGET_FILE_DIR:mapping_tests>/mapping probe/csgo.exe")
    foreach(scenario success corrupt foreign entry-false empty-entry lifecycle iat entry-thread missing)
        add_test(NAME mapping-${scenario} COMMAND mapping_tests ${scenario})
        set_tests_properties(mapping-${scenario} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
