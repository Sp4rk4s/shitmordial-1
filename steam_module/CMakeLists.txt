cmake_minimum_required(VERSION 3.20)
project(steam_module LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Build for the architecture of the parent steam.exe process.
set(PAYLOAD_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../primordial.bin")
configure_file(payload.rc.in "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" @ONLY)
add_library(${PROJECT_NAME} SHARED dllmain.cpp staging.cpp iat_hook.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_compile_definitions(${PROJECT_NAME} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
include(CTest)
set(MAPPING_TEST_PROBE "" CACHE FILEPATH "Path to the x86 mapping_tests executable copied as csgo.exe")
if(BUILD_TESTING AND MAPPING_TEST_PROBE)
    add_executable(steam_launch_tests ../tests/steam_launch_tests.cpp staging.cpp iat_hook.cpp)
    target_compile_definitions(steam_launch_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    foreach(scenario normal-w normal-a suspended command-only staging-failure unrelated)
        add_test(NAME steam-${scenario} COMMAND steam_launch_tests "${MAPPING_TEST_PROBE}" ${scenario})
        set_tests_properties(steam-${scenario} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
